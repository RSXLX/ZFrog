    prisma.crossChainTransfer.findMany({

      select: { amount: true },
    }),
  ]);
 
  // Calculate total volume manually (amount is a string)
  const totalVolume = transfers.reduce((sum: number, t: { amount: string }) => {
    const val = parseFloat(t.amount) || 0;
    return sum + val;

  return {
    sentCount: sent,
    receivedCount: received,
